{{/* responsive-image.html — the keystone partial. Emits a with WebP srcset over several widths + a JPEG fallback . Width/height are set on to prevent CLS; lazy by default (hero passes lazy=false). Usage: {{ partial "responsive-image.html" (dict "img" $r "alt" "..." "sizes" "(min-width:1024px) 50vw, 100vw") }} Optional: "widths" (slice), "lazy" (bool), "class" (on ), "imgclass" (on ). `img` may be a page resource (.Resources) or a global resource (resources.Get). */}} {{- $img := .img -}} {{- $alt := .alt | default "" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $widths := .widths | default (slice 480 800 1200 1600 2048) -}} {{- $lazy := .lazy | default true -}} {{- $class := .class | default "block" -}} {{- $imgclass := .imgclass | default "w-full h-auto" -}} {{- if not $img -}} {{/* graceful fallback so a missing resource never breaks the build */}}
[image missing]
{{- else -}} {{- $fallback := $img.Process (printf "resize %dx jpg q80" (index $widths 2)) -}} {{- range $w := $widths -}} {{- $r := $img.Process (printf "resize %dx webp q72" $w) -}} {{- end }} {{- end -}}
